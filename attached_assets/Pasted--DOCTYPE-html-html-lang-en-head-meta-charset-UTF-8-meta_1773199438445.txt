  <script>
    function esc(str) {
      return String(str)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }

    function getValues() {
      return {
        logoUrl: document.getElementById("logoUrl").value.trim(),
        fullName: document.getElementById("fullName").value.trim(),
        title: document.getElementById("title").value.trim(),
        company: document.getElementById("company").value.trim(),
        email: document.getElementById("email").value.trim(),
        website: document.getElementById("website").value.trim(),
        location: document.getElementById("location").value.trim()
      };
    }

    function buildSignatureHTML(data) {
      const websiteText = data.website.replace(/^https?:\/\//, "");
      return `
<table cellpadding="0" cellspacing="0" border="0" style="font-family: Arial, Helvetica, sans-serif; color: #222222; line-height: 1.2;">
  <tr>
    <td valign="top" style="padding-right: 22px;">
      <img src="${esc(data.logoUrl)}" alt="Hire'in Solutions Logo" width="170" style="display:block; width:170px; max-width:170px; height:auto; border:0;">
    </td>
    <td valign="top" style="padding-top: 2px;">
      <div style="font-size: 22px; font-weight: 700; color: #111111; margin: 0 0 6px 0;">
        ${esc(data.fullName)}
      </div>
      <div style="font-size: 18px; color: #444444; margin: 0 0 6px 0;">
        ${esc(data.title)}
      </div>
      <div style="font-size: 18px; font-weight: 700; color: #111111; margin: 0 0 12px 0;">
        ${esc(data.company)}
      </div>

      <div style="font-size: 16px; margin: 0 0 10px 0;">
        <span style="display:inline-block; width:24px;">&#128231;</span>
        <a href="mailto:${esc(data.email)}" style="color:#145cb3; text-decoration: underline;">${esc(data.email)}</a>
      </div>

      <div style="font-size: 16px; margin: 0 0 10px 0;">
        <span style="display:inline-block; width:24px;">&#127760;</span>
        <a href="${esc(data.website)}" style="color:#145cb3; text-decoration: underline;" target="_blank">${esc(websiteText)}</a>
      </div>

      <div style="font-size: 16px; margin: 0;">
        <span style="display:inline-block; width:24px;">&#128205;</span>
        <span style="color:#222222;">${esc(data.location)}</span>
      </div>
    </td>
  </tr>
</table>`.trim();
    }

    function renderSignature() {
      const data = getValues();
      const html = buildSignatureHTML(data);
      document.getElementById("preview").innerHTML = html;
      document.getElementById("htmlOutput").value = html;
    }

    async function copyHTML() {
      const html = document.getElementById("htmlOutput").value;
      try {
        await navigator.clipboard.writeText(html);
        alert("Signature HTML copied.");
      } catch (e) {
        alert("Copy failed. Please copy manually from the box.");
      }
    }

    function downloadHTML() {
      const html = document.getElementById("htmlOutput").value;
      const fullDoc = `<!DOCTYPE html>
<html lang="en">
<head><meta charset="UTF-8"><title>Hire'in Signature</title></head>
<body>${html}</body>
</html>`;
      const blob = new Blob([fullDoc], { type: "text/html" });
      const a = document.createElement("a");
      a.href = URL.createObjectURL(blob);
      a.download = "hirein-signature.html";
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
    }

    renderSignature();
  </script>
