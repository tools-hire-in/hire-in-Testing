CREATE TYPE "public"."salary_report_status" AS ENUM('pending_approval', 'approved', 'sent');
CREATE TABLE "salary_report_runs" (
	"id" varchar PRIMARY KEY DEFAULT gen_random_uuid() NOT NULL,
	"year" integer NOT NULL,
	"month" integer NOT NULL,
	"status" "salary_report_status" DEFAULT 'pending_approval' NOT NULL,
	"generated_at" timestamp DEFAULT now(),
	"approved_at" timestamp,
	"approved_by" varchar,
	"report_data" jsonb DEFAULT '[]'::jsonb NOT NULL,
	"adjustments" jsonb DEFAULT '{}'::jsonb NOT NULL,
	"email_sent_at" timestamp,
	"created_at" timestamp DEFAULT now()
);
CREATE TABLE "goal_milestones" (
	"id" varchar PRIMARY KEY DEFAULT gen_random_uuid() NOT NULL,
	"goal_id" varchar NOT NULL,
	"title" varchar NOT NULL,
	"target_date" varchar,
	"done" boolean DEFAULT false NOT NULL,
	"completed_at" timestamp,
	"sort_order" integer DEFAULT 0 NOT NULL,
	"created_at" timestamp DEFAULT now(),
	"updated_at" timestamp DEFAULT now()
);
