from docx import Document
from docx.shared import Pt
from docx.oxml import OxmlElement
from pathlib import Path

src = Path("/mnt/data/Hirein_Content_Insights_and_Social_Media_MVP_Strategy_v2.docx")
out = Path("/mnt/data/Hirein_Content_Insights_and_Social_Media_MVP_Strategy_v3.docx")

doc = Document(src)

def insert_paragraph_after(paragraph, text="", style=None):
    new_p = OxmlElement("w:p")
    paragraph._p.addnext(new_p)
    new_para = paragraph._parent.add_paragraph()
    new_para._p.getparent().remove(new_para._p)
    new_p.addnext(new_para._p)
    # Swap positions: place actual paragraph where placeholder is
    parent = new_p.getparent()
    idx = parent.index(new_p)
    parent.remove(new_p)
    parent.insert(idx, new_para._p)
    if style:
        new_para.style = style
    if text:
        new_para.add_run(text)
    return new_para

def insert_label_after(paragraph, label, body):
    p = insert_paragraph_after(paragraph)
    p.paragraph_format.space_after = Pt(4)
    r = p.add_run(label)
    r.bold = True
    p.add_run(body)
    return p

# 1. Audience translation note after Supporting brand layer paragraph.
target = next(p for p in doc.paragraphs if p.text.strip().startswith(
    "Process, people, company culture and Hire'in perspective"
))
p = insert_paragraph_after(target, "Audience labels in the strategy and Studio", "Heading 2")
p = insert_paragraph_after(
    p,
    "The H1, H2, I1 and I2 labels are planning shorthand used in this strategy. "
    "The Content Studio stores audience and staffing domain as separate selections. "
    "Use the following translation when creating content in the Studio:"
)
for text in [
    "H1 = EMPLOYER_CLIENT audience + HEALTHCARE domain.",
    "H2 = CANDIDATE audience + HEALTHCARE domain.",
    "I1 = EMPLOYER_CLIENT audience + IT domain.",
    "I2 = CANDIDATE audience + IT domain.",
]:
    p = insert_paragraph_after(p, text, "List Bullet")
p = insert_paragraph_after(
    p,
    "When the Studio displays additional audiences such as MSP/VMS or Staffing Partner and Recruiter or "
    "Staffing Operator, select them only when the content is specifically written for that audience. "
    "The audience shown in the Studio, the content brief and the saved article metadata must remain consistent."
)

# 2. Platform equivalent for People and Perspective.
target = next(p for p in doc.paragraphs if p.text.strip() == "Responsible recruiting practices.")
p = insert_paragraph_after(target, "Studio content-goal mapping", "Heading 3")
p = insert_paragraph_after(
    p,
    "In the Content Studio, select Brand Perspective for People and Perspective content. "
    "This includes founder or leadership points of view, recruiter insights, team learning, culture, "
    "candidate-care principles and responsible recruiting practices. Brand Perspective is marketing content "
    "that explains how Hire'in thinks; it is not a formal capability statement, proposal or business-development document."
)

# 3. Measurement source clarity after Measurement approach paragraph.
target = next(p for p in doc.paragraphs if p.text.strip().startswith(
    "The first 30 days are intended to establish a reliable baseline"
))
p = insert_paragraph_after(target, "Where each metric comes from", "Heading 2")
p = insert_label_after(
    p,
    "Native social-platform analytics: ",
    "LinkedIn, Instagram, Facebook and X provide reach or impressions, reactions, saves, shares, comments, "
    "video views and other platform-native engagement. These metrics are not generated by the Content Studio "
    "and must be recorded from the relevant social platform."
)
p = insert_label_after(
    p,
    "Content Studio and website analytics: ",
    "The Studio records the content brief, audience, domain, content goal, platform, publishing details and "
    "available CTA or link-click activity. Website analytics measure what happens after a reader follows a link, "
    "including article visits, referral traffic, page engagement and CTA clicks where tracking is configured."
)
p = insert_label_after(
    p,
    "ATS and recruiter records: ",
    "Applications, recruiter contacts, qualified candidate conversations, employer inquiries and other downstream "
    "actions should be confirmed through the ATS, recruiting records, shared inbox or responsible owner."
)
p = insert_label_after(
    p,
    "Manual MVP tracker: ",
    "Audience questions, qualitative feedback, production time, approval delays and successful content reuse should "
    "be recorded manually during the pilot."
)
p = insert_paragraph_after(
    p,
    "Studio data should not be presented as native LinkedIn, Instagram, Facebook or X reach. "
    "The 30-day review should combine the four sources above to produce one complete performance view."
)

doc.save(out)
print(f"Created: {out}")
print(f"Size: {out.stat().st_size:,} bytes")
