export interface ExperienceLetterFormState {
  documentType: string;
  roleKey: string;
  performanceBandKey: string;
  conductBandKey: string;
  includeResponsibilitiesSummary: boolean;
  summaryVariant: 'default' | 'alternate';
}

export const defaultExperienceLetterFormState: ExperienceLetterFormState = {
  documentType: 'employee_experience_letter',
  roleKey: '',
  performanceBandKey: 'standard',
  conductBandKey: 'satisfactory',
  includeResponsibilitiesSummary: true,
  summaryVariant: 'default',
};

export function getActiveDocumentTypes() {
  return documentTypes.filter((item) => item.isActive);
}

export function getActiveRoleSummaries() {
  return roleSummaryLibrary
    .filter((item) => item.isActive)
    .sort((a, b) => a.sortOrder - b.sortOrder);
}

export function getRoleSummaryByKey(roleKey: string) {
  return roleSummaryLibrary.find((item) => item.roleKey === roleKey) ?? null;
}

export function getPerformanceBandByKey(key: string) {
  return performanceBands.find((item) => item.key === key) ?? null;
}

export function getConductBandByKey(key: string) {
  return conductBands.find((item) => item.key === key) ?? null;
}

export function getRoleOptionsForDropdown() {
  return getActiveRoleSummaries().map((item) => ({
    label: item.roleLabel,
    value: item.roleKey,
  }));
}

export function getResponsibilityOptionsForRole(roleKey: string) {
  const role = getRoleSummaryByKey(roleKey);
  if (!role) return [];

  const options = [
    { label: 'Default Summary', value: 'default', text: role.defaultSummary },
  ];

  if (role.alternateSummary) {
    options.push({
      label: 'Alternate Summary',
      value: 'alternate',
      text: role.alternateSummary,
    });
  }

  return options;
}

export function getSelectedResponsibilityText(
  roleKey: string,
  variant: 'default' | 'alternate' = 'default',
) {
  const role = getRoleSummaryByKey(roleKey);
  if (!role) return '';

  if (variant === 'alternate' && role.alternateSummary) {
    return role.alternateSummary;
  }

  return role.defaultSummary;
}

export function buildExperienceLetterParagraphs(params: {
  fullName: string;
  employeeId?: string;
  companyName: string;
  designation: string;
  department: string;
  startDate: string;
  endDate: string;
  roleKey: string;
  summaryVariant?: 'default' | 'alternate';
  includeResponsibilitiesSummary?: boolean;
  performanceBandKey: string;
  conductBandKey: string;
}) {
  const {
    fullName,
    employeeId,
    companyName,
    designation,
    department,
    startDate,
    endDate,
    roleKey,
    summaryVariant = 'default',
    includeResponsibilitiesSummary = true,
    performanceBandKey,
    conductBandKey,
  } = params;

  const responsibilities = includeResponsibilitiesSummary
    ? getSelectedResponsibilityText(roleKey, summaryVariant)
    : '';

  const performanceLine = getPerformanceBandByKey(performanceBandKey)?.line ??
    getPerformanceBandByKey('standard')?.line ?? '';

  const conductLine = getConductBandByKey(conductBandKey)?.line ??
    getConductBandByKey('satisfactory')?.line ?? '';

  const intro = `This is to certify that ${fullName}${employeeId ? `, Employee ID ${employeeId},` : ''} was employed with ${companyName} as ${designation} in the ${department} department from ${startDate} to ${endDate}.`;

  const responsibilitiesLine = includeResponsibilitiesSummary && responsibilities
    ? responsibilities
    : '';

  return {
    intro,
    responsibilitiesLine,
    performanceLine,
    conductLine,
    closingLine: 'We wish the individual success in future endeavors.',
  };
}
